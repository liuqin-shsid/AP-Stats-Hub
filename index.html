  <!-- 不修改 scatterplot.js 的前提下，让 samplingPage 能切换显示 -->

  <script>

    (function () {

      function activate(tool) {

        var pages = ['correlationPage','scatterPage','outlierPage','linearPage','samplingPage'];

        pages.forEach(function (id) {

          var el = document.getElementById(id);

          if (el) el.hidden = (id !== tool + 'Page');

        });

        document.querySelectorAll('[data-tool]').forEach(function (btn) {

          var active = btn.dataset.tool === tool;

          btn.classList.toggle('active', active);

          if (active) btn.setAttribute('aria-current','page');

          else btn.removeAttribute('aria-current');

        });

        var map = { correlation:'heading_title', scatter:'scatter_title', outlier:'outlier_title', linear:'linear_title', sampling:'sampling_title' };

        try {

          if (typeof translations !== 'undefined' && typeof lang !== 'undefined') {

            document.title = 'AP Stats Hub · ' + (translations[lang][map[tool]] || '');

          }

        } catch (e) {}

        document.dispatchEvent(new CustomEvent('apstats:tool', { detail: tool }));

      }


      document.addEventListener('DOMContentLoaded', function () {

        document.querySelectorAll('[data-tool]').forEach(function (btn) {

          btn.addEventListener('click', function () { activate(btn.dataset.tool); });

        });

      });

    })();

  </script>
